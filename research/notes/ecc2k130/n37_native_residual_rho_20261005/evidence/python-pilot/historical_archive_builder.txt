#!/usr/bin/env python3
"""Seal the complete raw panel as a deterministic tar.gz with file hashes."""

import argparse
import gzip
import hashlib
import json
import tarfile
from pathlib import Path


def digest(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--raw-dir", required=True, type=Path)
    parser.add_argument("--archive", required=True, type=Path)
    parser.add_argument("--manifest", required=True, type=Path)
    args = parser.parse_args()
    assert args.raw_dir.is_dir()
    assert not args.archive.exists() and not args.manifest.exists()
    files = sorted(path for path in args.raw_dir.rglob("*") if path.is_file())
    assert len(files) >= 60
    args.archive.parent.mkdir(parents=True, exist_ok=True)
    with args.archive.open("xb") as output:
        with gzip.GzipFile(fileobj=output, mode="wb", filename="", mtime=0,
                           compresslevel=9) as compressed:
            with tarfile.open(fileobj=compressed, mode="w", format=tarfile.USTAR_FORMAT) as tar:
                for path in files:
                    name = "raw-panel/" + path.relative_to(args.raw_dir).as_posix()
                    info = tarfile.TarInfo(name)
                    info.size = path.stat().st_size
                    info.mode = 0o644
                    info.uid = info.gid = info.mtime = 0
                    info.uname = info.gname = ""
                    with path.open("rb") as source:
                        tar.addfile(info, source)
    receipt = {
        "schema": "n37-native-batch-rho-archive-v1",
        "archive_name": args.archive.name,
        "archive_sha256": digest(args.archive),
        "archive_bytes": args.archive.stat().st_size,
        "files": [{"path": path.relative_to(args.raw_dir).as_posix(),
                   "sha256": digest(path), "bytes": path.stat().st_size}
                  for path in files],
    }
    args.manifest.write_text(json.dumps(receipt, indent=2, sort_keys=True) + "\n")
    with tarfile.open(args.archive, "r:gz") as tar:
        assert len(tar.getmembers()) == len(files)
        for row in receipt["files"]:
            entry = tar.getmember("raw-panel/" + row["path"])
            content = tar.extractfile(entry).read()
            assert len(content) == row["bytes"]
            assert hashlib.sha256(content).hexdigest() == row["sha256"]
    print(json.dumps({"archive_sha256": receipt["archive_sha256"],
                      "archive_bytes": receipt["archive_bytes"],
                      "file_count": len(files)}))


if __name__ == "__main__":
    main()
